MODULE 06

🔄 JavaScript Loops & Functions

Learn how to repeat tasks and create reusable JavaScript code.

TOPIC 01

Why Do We Need Loops?

Sometimes a program needs to perform the same action many times. A loop helps us avoid writing the same code repeatedly.

TOPIC 02

The for Loop

for (let i = 1; i <= 5; i++) {

    console.log(i);

}

This example prints numbers from 1 to 5.

TOPIC 03

Looping Through an Array

let courses = [
    "HTML",
    "CSS",
    "JavaScript"
];

for (let i = 0; i < courses.length; i++) {

    console.log(courses[i]);

}

TOPIC 04

for...of

let courses = [
    "HTML",
    "CSS",
    "JavaScript"
];

for (let course of courses) {

    console.log(course);

}

TOPIC 05

Functions Review

function welcome() {

    console.log("Welcome to my website!");

}

welcome();

TOPIC 06

Functions With Parameters

function greet(name) {

    console.log("Hello " + name);

}

greet("Neema");

TOPIC 07

Returning a Value

function add(a, b) {

    return a + b;

}

let result = add(10, 20);

console.log(result);

TOPIC 08

Combining Functions and Loops

function showCourses(courses) {

    for (let course of courses) {

        console.log(course);

    }

}

showCourses([
    "HTML",
    "CSS",
    "JavaScript"
]);
